cmake_minimum_required(VERSION 3.21)
project(dai_alignment LANGUAGES CXX)

find_package(nlohmann_json 3.11 REQUIRED CONFIG)
find_package(Eigen3 3.4 REQUIRED NO_MODULE)
find_package(TinyGLTF REQUIRED CONFIG)

add_library(alignment_core STATIC src/alignment.cpp src/inspection.cpp)
target_include_directories(alignment_core PUBLIC include)
target_compile_features(alignment_core PUBLIC cxx_std_20)
target_link_libraries(alignment_core PUBLIC nlohmann_json::nlohmann_json PRIVATE Eigen3::Eigen tinygltf::tinygltf)
add_executable(dai-alignment-worker src/main.cpp)
target_link_libraries(dai-alignment-worker PRIVATE alignment_core nlohmann_json::nlohmann_json)
foreach(target alignment_core dai-alignment-worker)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /utf-8 /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
if(APPLE)
  set_target_properties(dai-alignment-worker PROPERTIES INSTALL_RPATH "@loader_path/lib")
elseif(UNIX)
  set_target_properties(dai-alignment-worker PROPERTIES INSTALL_RPATH "$ORIGIN/lib")
endif()
install(TARGETS dai-alignment-worker RUNTIME_DEPENDENCY_SET builtin_runtime RUNTIME DESTINATION .)
